if __name__ == '__main__':
    N = int(input())

    arr = []

    for i in range(N):
        data = input().split()
        command = data[0]

        if command == "insert":
            arr.insert(int(data[1]), int(data[2]))
        elif command == "print":
            print(arr)
        elif command == "remove":
            arr.remove(int(data[1]))
        elif command == "append":
            arr.append(int(data[1]))
        elif command == "sort":
            arr.sort()
        elif command == "pop":
            arr.pop()
        elif command == "reverse":
            arr.reverse()
